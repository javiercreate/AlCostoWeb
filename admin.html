🛠️ Script SQL para Crear las Tablas en Supabase (Copiar y Pegar en Supabase SQL Editor)

Si creaste un proyecto nuevo en Supabase, ve a SQL Editor, pega este código y pulsa Run. Creará las tablas productos y pedidos con permisos de lectura y escritura abiertos. 

-- 1. Tabla de Productos
CREATE TABLE IF NOT EXISTS productos (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  category TEXT NOT NULL,
  price NUMERIC NOT NULL,
  original_price NUMERIC,
  wholesale_price NUMERIC,
  wholesale_min INTEGER DEFAULT 1,
  unit TEXT,
  image TEXT,
  badge TEXT,
  description TEXT,
  origin TEXT,
  is_wholesale BOOLEAN DEFAULT FALSE,
  in_stock BOOLEAN DEFAULT TRUE,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. Tabla de Pedidos / Compras
CREATE TABLE IF NOT EXISTS pedidos (
  id TEXT PRIMARY KEY,
  customer_name TEXT NOT NULL,
  customer_phone TEXT NOT NULL,
  delivery_address TEXT NOT NULL,
  delivery_slot TEXT,
  instructions TEXT,
  payment_method TEXT,
  items JSONB NOT NULL,
  subtotal NUMERIC NOT NULL,
  discount NUMERIC DEFAULT 0,
  delivery_fee NUMERIC DEFAULT 0,
  tip NUMERIC DEFAULT 0,
  total NUMERIC NOT NULL,
  status TEXT DEFAULT 'nuevo',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Habilitar políticas públicas para lectura y creación (anon)
ALTER TABLE productos ENABLE ROW LEVEL SECURITY;
ALTER TABLE pedidos ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Lectura pública de productos" ON productos FOR SELECT USING (true);
CREATE POLICY "Gestión pública de productos" ON productos FOR ALL USING (true);

CREATE POLICY "Lectura pública de pedidos" ON pedidos FOR SELECT USING (true);
CREATE POLICY "Creación pública de pedidos" ON pedidos FOR INSERT WITH CHECK (true);
CREATE POLICY "Actualización de pedidos" ON pedidos FOR UPDATE USING (true);
